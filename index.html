Production Flask Backend & DB Blueprint

Below is the modular production Flask application code and database model designed to serve this frontend UI using SQLite / PostgreSQL:

# app.py - Flask Trading Journal Backend
from flask import Flask, jsonify, request
from flask_sqlalchemy import SQLAlchemy
from datetime import datetime

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///trading_journal.db'
db = SQLAlchemy(app)

class Trade(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    symbol = db.Column(db.String(20), nullable=False)
    market = db.Column(db.String(20), nullable=False)
    strategy = db.Column(db.String(50))
    entry_price = db.Column(db.Float, nullable=False)
    exit_price = db.Column(db.Float, nullable=False)
    stop_loss = db.Column(db.Float, nullable=False)
    size = db.Column(db.Integer, nullable=False)
    pnl = db.Column(db.Float, nullable=False)
    r_multiple = db.Column(db.Float, nullable=False)
    psych_state = db.Column(db.String(50))
    mistake_tag = db.Column(db.String(50))
    notes = db.Column(db.Text)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)

@app.route('/api/trades', methods=['GET'])
def get_trades():
    trades = Trade.query.order_by(Trade.created_at.desc()).all()
    return jsonify([{
        'id': t.id, 'symbol': t.symbol, 'market': t.market,
        'strategy': t.strategy, 'pnl': t.pnl, 'r_multiple': t.r_multiple
    } for t in trades])

@app.route('/api/trades', methods=['POST'])
def add_trade():
    data = request.json
    trade = Trade(**data)
    db.session.add(trade)
    db.session.commit()
    return jsonify({"status": "success", "id": trade.id}), 201

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True, port=5000)
